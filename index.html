🛠️ Struktur SQL Tabel `public.orders` (Supabase)

Jalankan di SQL Editor Supabase jika ada kolom yang belum dibuat

create table if not exists public.orders (
  id bigint generated by default as identity primary key,
  created_at timestamptz default now(),
  initials text,
  product_name text,
  plan_label text,
  base_price numeric default 0,
  discount_pct numeric default 0,
  saved_amount numeric default 0,
  final_price numeric default 0,
  status text default 'WAITING_QRIS',
  qris_image_url text,
  feedback text
);

alter table public.orders enable row level security;
create policy "Allow public read/write orders" on public.orders
  for all using (true) with check (true);
